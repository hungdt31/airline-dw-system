% --- Insight & khuyến nghị ---
\begin{frame}{Insight thực tế \& khuyến nghị (dữ liệu 2015)}
Kết hợp OLAP dashboard và Feature Importance:

\begin{enumerate}\setlength\itemsep{0.25em}
  \item \textbf{Dây chuyền (Late Aircraft + NAS):} tỷ trọng trễ lớn --- trễ sáng có thể lan 3--5 chuyến.\\
        {\footnotesize\textit{Khuyến nghị:} tăng buffer quay đầu (\textit{turnaround}) tại hub trung chuyển.}

  \item \textbf{Hai đỉnh delay:} mùa hè (T6--T7) và mùa đông (T12--T1) --- trùng bão/tuyết.\\
        {\footnotesize\textit{Khuyến nghị:} phương án vận hành riêng mùa cao điểm (nhân lực mặt đất, lịch bay).}

  \item \textbf{Chuyến trước 8h:} ít trễ nhất (chưa tích lũy trong ngày).\\
        {\footnotesize\textit{Khuyến nghị hành khách:} ưu tiên chuyến sáng nếu cần nối chuyến quan trọng.}

  \item \textbf{Sân bay là nút thắt:} ATL vs ORD minh họa điều phối/hạ tầng; NAS $\sim$50\% ca trễ.\\
        {\footnotesize\textit{Khuyến nghị:} minh bạch năng lực giải tỏa, điều phối slot --- cắt chuỗi trễ dây chuyền.}

  \item \textbf{Quãng đường + thời gian bay} quan trọng trong mô hình; \textbf{bang} (\texttt{origin}/\texttt{dest}) phản ánh địa lý/khí hậu.
\end{enumerate}
\end{frame}

% --- Hạn chế ---
\begin{frame}{Hạn chế}

\begin{enumerate}
  \item \textbf{Dữ liệu chỉ năm 2015.} Insight/mô hình là ``ảnh chụp'' một năm; hạ tầng, FAA, hành vi sau COVID có thể khác --- \textit{phương pháp luận} (DW + pipeline ML) vẫn tái sử dụng khi đổi nguồn.

  \item \textbf{SQLite không phù hợp production.} Single-writer, không concurrent; OLAP nặng vài giây --- đủ demo, chưa đủ đa người dùng.

  \item \textbf{AUC-ROC $\sim$ 0{,}76.} Cố tình \textbf{không} dùng \texttt{departure\_delay} (dự báo thật sự). Thiếu thời tiết thực, bảo trì, lịch sử theo \texttt{tail\_number} --- có thể cải thiện thêm khoảng 0{,}03--0{,}10 AUC.

  \item \textbf{Chưa có Clustering / Association Rules.} Tập trung classification + OLAP; có thể bổ sung phân cụm sân bay hoặc luật kết hợp (vd.\ hãng + sân bay ORD + T12 $\Rightarrow$ trễ).
\end{enumerate}
\end{frame}

% --- Hướng phát triển ---
\begin{frame}{Hướng phát triển}

\begin{enumerate}
  \item \textbf{Dữ liệu nhiều năm (BTS 2010--2023):} xu hướng dài hạn, tác động COVID (2020--2021), mô hình tổng quát hơn.

  \item \textbf{Thời tiết NOAA} theo sân bay + ngày --- yếu tố lớn mà mô hình hiện chưa có.

  \item \textbf{Nâng cấp hạ tầng:} PostgreSQL / ClickHouse; Docker + cloud (EC2 / Cloud Run); \textbf{REST API} (FastAPI) phục vụ dự báo theo yêu cầu.

  \item \textbf{SHAP:} giải thích \textit{từng} dự báo --- dashboard không chỉ ``có trễ'' mà ``vì sao trễ''.

  \item \textbf{Streaming thời gian thực:} Kafka nhận chuyến bay real-time + Spark Streaming --- từ phân tích lịch sử sang hỗ trợ quyết định tức thời.
\end{enumerate}
\end{frame}
