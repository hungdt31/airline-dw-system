\begin{frame}{Tri thức khai phá \& Đề xuất vận hành}
  \begin{enumerate}
    \item \textbf{Hiệu ứng Domino (Late Aircraft):} Trễ chuyến có tính chất lây lan mạng lưới. \textbf{Khuyến nghị:} Tăng thời gian đệm (\textit{Buffer}) tại các khung giờ cao điểm để ngắt chuỗi trễ dây chuyền.
    \item \textbf{Tính mùa vụ rõ rệt:} Đỉnh điểm trễ chuyến rơi vào mùa hè và mùa đông do áp lực vận hành và thời tiết. \textbf{Khuyến nghị:} Tối ưu hóa nguồn lực nhân sự mặt đất theo chu kỳ mùa.
    \item \textbf{Ưu thế khung giờ sáng:} Các chuyến trước 08:00 có tỷ lệ đúng giờ cao nhất. \textbf{Khuyến nghị hành khách:} Ưu tiên bay sớm cho các hành trình có nối chuyến phức tạp.
    \item \textbf{Nút thắt hạ tầng (NAS):} Các yếu tố điều phối sân bay chiếm $\sim$50\% nguyên nhân trễ. \textbf{Khuyến nghị:} Minh bạch năng lực điều phối Slot để giảm tải áp lực cho các Hub trung chuyển lớn.
  \end{enumerate}
\end{frame}

\begin{frame}{Giới hạn của nghiên cứu}
  \begin{itemize}
    \item \textbf{Tính thời điểm của dữ liệu:} Dữ liệu năm 2015 mang tính chất "ảnh chụp" lịch sử. Tuy nhiên, \textbf{phương pháp luận} về xây dựng Kho dữ liệu và Pipeline ML hoàn toàn có thể tái sử dụng cho các tập dữ liệu mới hơn.
    \item \textbf{Hiệu năng hạ tầng:} Sử dụng SQLite phù hợp cho giai đoạn thử nghiệm (PoC) nhưng sẽ gặp rào cản về khả năng truy cập đồng thời trong môi trường vận hành thực tế.
    \item \textbf{Ranh giới dự báo:} Việc không sử dụng các giá trị delay thực tế giúp dự báo có giá trị thực tiễn nhưng cũng đặt ra thách thức lớn về độ chính xác khi thiếu các biến số về thời tiết thực tế và lịch sử bảo trì tàu bay.
    \item \textbf{Phạm vi phương pháp luận:} Tập trung chủ yếu vào bài toán phân lớp và trực quan hóa. Chưa ứng dụng các kỹ thuật \textbf{Clustering} (phân nhóm sân bay/tuyến đường) và \textbf{Association Rules} để nhận diện các mô thức trễ chuyến đặc thù gắn liền với các điều kiện vận hành phức tạp.
  \end{itemize}
\end{frame}

\begin{frame}{Tầm nhìn \& Định hướng phát triển}
  Hệ thống được thiết kế với khả năng mở rộng linh hoạt, hướng tới một nền tảng phân tích toàn diện và sẵn sàng cho môi trường vận hành thực tế:
  
  \begin{enumerate}
    \item \textbf{Mở rộng quy mô và tính tổng quát của dữ liệu:}
    \begin{itemize}
      \item \textbf{Tích hợp dữ liệu nhiều năm (2010--2025):} Phân tích biến động dài hạn và tác động hậu đại dịch.
      \item \textbf{Làm giàu đặc trưng (Feature Engineering):} Kết hợp dữ liệu thời tiết thực tế từ NOAA để tối ưu hóa độ chính xác của mô hình.
    \end{itemize}

    \item \textbf{Hiện đại hóa hạ tầng \& Chuyển dịch Cloud-native:}
    \begin{itemize}
      \item Nâng cấp hệ quản trị dữ liệu sang PostgreSQL hoặc ClickHouse để tối ưu hóa hiệu năng OLAP.
      \item Triển khai giải pháp \textbf{FastAPI} trên nền tảng Docker/Cloud.
    \end{itemize}

    \item \textbf{Nâng cao năng lực phân tích thông minh:}
    \begin{itemize}
      \item \textbf{Explainable AI (SHAP):} Minh bạch hóa các quyết định của mô hình, giúp Dashboard không chỉ dự báo "trễ" mà còn giải thích được "tại sao trễ".
      \item \textbf{Xử lý dòng thời gian thực (Real-time Pipeline):} Ứng dụng Kafka và Spark Streaming để chuyển đổi từ phân tích lịch sử sang hỗ trợ quyết định tức thời.
    \end{itemize}
  \end{enumerate}
\end{frame}
