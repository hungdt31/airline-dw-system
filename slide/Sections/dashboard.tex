\section{Phân tích và trình bày}


\begin{frame}{Phân tích OLAP đa chiều}
  \textbf{OLAP (Online Analytical Processing)} là kỹ thuật phân tích đa chiều, hỗ trợ khám phá tri thức thông qua việc tương tác và đào sâu dữ liệu linh hoạt trên Star Schema.

  \vspace{2mm}
  \begin{columns}[T,onlytextwidth]
      \begin{column}{0.48\textwidth}
          \begin{block}{5 thao tác tương tác cốt lõi}
              \begin{description}[\textbf{Drill-down}]
                  \item[\textbf{Roll-up}] Tổng hợp mức cao (Tháng $\rightarrow$ Quý).
                  \item[\textbf{Drill-down}] Chi tiết hóa (Hãng $\rightarrow$ Chuyến).
                  \item[\textbf{Slice}] Cắt lớp theo 1 chiều (Tháng 12).
                  \item[\textbf{Dice}] Lọc theo nhiều chiều (Hãng A + Bang B).
                  \item[\textbf{Pivot}] Xoay trục quan sát (Hãng bay $\leftrightarrow$ Sân bay).
              \end{description}
          \end{block}
      \end{column}
      
      \begin{column}{0.48\textwidth}
          \begin{block}{Hiện thực hóa trên Dashboard}
              \begin{itemize}
                  \item \textbf{Slice/Dice:} Tích hợp qua bộ lọc Sidebar trên toàn hệ thống.
                  \item \textbf{Roll-up/Drill-down:} Chuyển đổi linh hoạt giữa các cấp độ thời gian.
                  \item \textbf{Pivot:} Thay đổi góc nhìn dữ liệu trên các trang chuyên biệt.
              \end{itemize}
          \end{block}
      \end{column}
  \end{columns}
\end{frame}

\begin{frame}[shrink=8,t]{Bản chất của sự chậm trễ --- tổng phút trễ theo nguyên nhân}
\vspace{-1mm}
\centering
\includegraphics[width=\linewidth,height=0.5\textheight,keepaspectratio]{delay_minute.png}
\vspace{1mm}
\footnotesize
\begin{itemize}\setlength\itemsep{0.12em}
  \item \textbf{Tổng phút trễ:} máy bay đến muộn là ``nút thắt'' lớn.
        Trên biểu đồ tổng phút trễ, \textit{late aircraft} đứng thứ hai về tần suất (khoảng 30\%) nhưng đóng góp nhiều phút trễ nhất toàn hệ thống (gần 15 triệu phút).
  \item \textbf{Nhận định:} minh họa \textit{hiệu ứng hòn tuyết lăn} đã nêu ở khung giờ tối --- một chuyến trễ ban sáng ``lây lan'' cho các chuyến sau; tổng thời gian cộng dồn rất lớn.
  \item \textbf{Thời tiết:} chỉ khoảng 6{,}14\% tổng số ca trễ, nhưng thời gian chờ TB gần 48 phút.
  \item \textbf{Nhận định:} thời tiết không thường xuyên trên toàn mạng, nhưng khách quan, khó kiểm soát và thường gây gián đoạn đáng kể cho mỗi chuyến bị ảnh hưởng.
\end{itemize}
