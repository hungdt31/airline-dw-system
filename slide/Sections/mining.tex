\section{Khai phá dữ liệu}

% --- Slide: Phát biểu bài toán ---
\begin{frame}{Phát biểu bài toán}
\begin{block}{Câu hỏi khai phá}
\textit{``Có thể dự đoán một chuyến bay sẽ trễ đến $\geq 15$ phút hay không,
chỉ dựa trên thông tin biết \textbf{trước} khi cất cánh?''}
\end{block}

\vspace{2mm}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.52\textwidth}
\small\textbf{Phân lớp nhị phân}
\begin{itemize}\setlength\itemsep{0.2em}
  \item \textbf{Nhãn:} \texttt{is\_delayed\_15} --- 1 nếu \texttt{arrival\_delay} $\geq 15$ phút, 0 nếu đúng giờ.
  \item \textbf{Mất cân bằng:} $\approx$\textbf{81{,}55\%} đúng giờ / \textbf{18{,}45\%} trễ (chuyến không hủy).
  \item \textbf{Không dùng} \texttt{departure\_delay} --- chỉ biết \textit{sau} khi rời cổng (phản ứng, không dự báo).
\end{itemize}
\end{column}
\begin{column}{0.48\textwidth}
\small\textbf{Quy trình end-to-end}
\begin{enumerate}\setlength\itemsep{0.15em}
  \item Lấy 500K mẫu (stratified theo tháng).
  \item Tiền xử lý + 10 đặc trưng.
  \item Holdout 80/20 (\texttt{stratify=y}).
  \item Huấn luyện 5 mô hình.
  \item Đánh giá 7 chỉ số + Confusion Matrix + Feature Importance.
\end{enumerate}
\end{column}
\end{columns}
\end{frame}

% --- Chuẩn bị dữ liệu ---
\begin{frame}[shrink=5]{Chuẩn bị dữ liệu: lấy mẫu \& tiền xử lý}
\small
\textbf{a. Lấy mẫu phân tầng theo tháng} (500.000 mẫu)
\begin{itemize}\setlength\itemsep{0.15em}
  \item Mỗi tháng $\approx \lfloor 500{.}000/12\rfloor = 41{.}666$ mẫu; phần dư (8) gán các tháng đầu $\to$ đủ 500.000.
  \item Trong tháng: \texttt{ORDER BY RANDOM()} từ \texttt{fact}; ghép 12 chunk rồi shuffle (\texttt{random\_state=42}).
  \item Giữ \textbf{tính mùa vụ} (bão hè, tuyết đông, lễ) thay vì lấy tuần tự từ đầu file.
\end{itemize}

\vspace{1mm}
\textbf{b. Tiền xử lý} (4 bước)
\begin{enumerate}\setlength\itemsep{0.12em}
  \item Loại dòng thiếu nhãn hoặc khóa: \texttt{target}, \texttt{month}, \texttt{day\_of\_week}, \texttt{airline\_iata}.
  \item \textbf{Median} cho \texttt{distance}, \texttt{elapsed\_time} khi thiếu.
  \item \texttt{dep\_hour} $\in [0,23]$; thiếu $\to$ 12 (giờ trưa).
  \item \textbf{Label Encoding:} \texttt{airline\_iata}$\to$\texttt{airline\_enc},
        \texttt{origin\_state}$\to$\texttt{origin\_enc}, \texttt{dest\_state}$\to$\texttt{dest\_enc}.
\end{enumerate}

\vspace{1mm}
\footnotesize\textit{LR \& MLP: thêm \texttt{StandardScaler}; RF / XGBoost / LightGBM: dữ liệu gốc.}
\end{frame}

% --- Tổng hợp GBDT ---
\begin{frame}{Tại sao Gradient Boosting tốt nhất?}
\small
\begin{enumerate}\setlength\itemsep{0.25em}
  \item \textbf{Quan hệ phi tuyến} --- LR không đủ biên quyết định tuyến tính; GBDT chia vùng dữ liệu.
  \item \textbf{Interaction tự học} --- ví dụ hãng $\times$ tháng $\times$ hub (không cần feature tay).
  \item \textbf{Boosting vs Bagging} --- cây sau sửa residual cây trước; RF vote trung bình độc lập.
  \item \textbf{Cân bằng lớp} --- \texttt{scale\_pos\_weight}, \texttt{is\_unbalance} chỉnh gradient.
  \item \textbf{LightGBM:} leaf-wise + histogram $\to$ \textbf{3{,}0s} vs.\ \textbf{6{,}7s} (XGB), AUC chỉ kém \textbf{0{,}0034} (0{,}7519 vs.\ 0{,}7553).
\end{enumerate}
\end{frame}
