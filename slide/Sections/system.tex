\begin{frame}{Nguyên tắc luồng dữ  liệu}
  \begin{itemize}
      \item \textbf{Luồng dữ liệu Một chiều (Unidirectional Flow)} \\
      Mô hình: \texttt{Raw Data} $\rightarrow$ \texttt{ETL} $\rightarrow$ \texttt{Data Warehouse}. \\
      Đảm bảo tính \textbf{tái lập}: Mọi trạng thái dữ liệu có thể khôi phục bằng cách chạy lại pipeline.
  
      \item \textbf{Nguyên tắc "Read-only" đối với Kho dữ liệu} \\
      Dashboard và Mining chỉ có quyền truy xuất (\textbf{Read}), không có quyền ghi. \\
      Giúp duy trì \textbf{Single Source of Truth}, bảo vệ tính toàn vẹn của Schema.
  
      \item \textbf{Tích hợp Liên thông ở Tầng Ứng dụng} \\
      Dashboard kết nối trực tiếp với kết quả từ Mining thay vì ghi ngược vào kho. Tránh gây "nhiễu" cấu trúc dữ liệu nền tảng.
  \end{itemize}
  
  \vspace{2mm}
  \begin{alertblock}{Giá trị cốt lõi: Tính Độc lập (Decoupling)}
      Cho phép phát triển song song: Cập nhật logic Dashboard hoặc nâng cấp thuật toán Mining mà không cần tái cấu trúc toàn bộ hệ thống ETL hay Schema kho.
  \end{alertblock}
  \end{frame}

\begin{frame}{Kiến trúc hệ thống ETL: Từ Dữ liệu thô đến Kho lưu trữ}
  Quy trình được thiết kế theo mô hình xử lý dòng (Streaming) nhằm tối ưu hóa tài nguyên RAM và đảm bảo tốc độ xử lý dữ liệu lớn.
  
  \begin{itemize}
      \item \textbf{Extract (Trích xuất):} Đọc dữ liệu thô từ CSV theo từng khối (Chunks), tối ưu hóa kiểu dữ liệu ngay từ bước nạp.
      \item \textbf{Transform (Biến đổi):} Chuẩn hóa mã định danh, xử lý dữ liệu khuyết thiếu và tính toán vector hóa.
      \item \textbf{Load (Nạp dữ liệu):} Chuyển đổi dữ liệu vào mô hình Star Schema trong SQLite với cơ chế ghi Batch.
  \end{itemize}
  
  \begin{block}{Hiệu quả hệ thống}
      \begin{itemize}
          \item Xử lý hàng triệu bản ghi trong 3--8 phút.
          \item Duy trì mức sử dụng RAM ổn định (~120MB) nhờ cơ chế Generator.
      \end{itemize}
  \end{block}
  \end{frame}

\begin{frame}{Bước 1: Extract - Trích xuất và Tối ưu bộ nhớ}
  Giai đoạn này không chỉ đơn thuần là đọc file mà là một bộ lọc thông minh giúp bảo vệ tài nguyên hệ thống:
  
  \begin{itemize}
      \item \textbf{Cơ chế Streaming (Generator):} Sử dụng tham số \texttt{chunksize=100,000} để chia nhỏ dữ liệu. Dữ liệu được đọc và xả khỏi RAM theo từng khối, ngăn chặn tình trạng tràn bộ nhớ.
      \item \textbf{Kỹ thuật Downcasting:} Ép kiểu dữ liệu tường minh (\texttt{int8}, \texttt{int16}, \texttt{float32}) ngay khi đọc. Kỹ thuật này giúp tiết kiệm khoảng \textbf{60\% dung lượng RAM} trên mỗi khối xử lý.
      \item \textbf{Lọc dữ liệu tại nguồn (Selective Reading):}
          \begin{itemize}
              \item Chỉ trích xuất các cột cần thiết (\texttt{USECOLS}).
              \item Loại bỏ ngay các bản ghi thiếu thông tin định danh quan trọng (\texttt{AIRLINE}, \texttt{ORIGIN\_AIRPORT}, \texttt{DESTINATION\_AIRPORT}).
          \end{itemize}
      \item \textbf{Quét danh mục độc lập:} Thực hiện một lượt quét riêng (\texttt{scan\_dim\_date}) để thu thập các giá trị thời gian duy nhất cho bảng Dimension.
  \end{itemize}
  \end{frame}

\begin{frame}{Bước 2: Transform - Tối ưu hóa Vector \& Logic nghiệp vụ}
  Giai đoạn biến đổi dữ liệu được thiết kế tối ưu hóa cho hiệu suất cực cao (tăng 50-100x) nhờ tư duy xử lý song song trên mảng:
  
  \begin{itemize}
      \item \textbf{Chiến lược Vector hóa (No-Loops):}
          \begin{itemize}
              \item Sử dụng \texttt{numpy.select} thay cho \texttt{if-else} để phân loại nguyên nhân trễ chuyến.
              \item Tính toán đồng thời trên toàn bộ cột dữ liệu, loại bỏ hoàn toàn vòng lặp \texttt{for}.
          \end{itemize}
      \item \textbf{Làm giàu dữ liệu thời gian (Rich Date Dimension):}
          \begin{itemize}
              \item Chuyển đổi Ngày/Tháng/Năm thành 11 thuộc tính phân tích (Quý, Tuần, Thứ, Cờ cuối tuần...).
              \item Tự động sinh khóa \texttt{date\_key} (YYYYMMDD) để tối ưu phép JOIN.
          \end{itemize}
      \item \textbf{Kỹ thuật đặc trưng (Feature Engineering):}
          \begin{itemize}
              \item Tạo các biến phân loại: Chuyến bay trễ ($>15$p), Bị hủy, Bị chuyển hướng.
              \item Chuẩn hóa mã sân bay: Giải quyết xung đột giữa mã số (BTS) và mã chữ (IATA).
          \end{itemize}
      \item \textbf{Bảo vệ tính toàn vẹn (Integrity):} Loại bỏ triệt để các bản ghi không khớp khóa ngoại (Foreign Keys) trước khi nạp vào kho.
  \end{itemize}
  \end{frame}

\begin{frame}{Bước 3: Load - Hiện thực hóa Kho dữ liệu}
  Giai đoạn cuối cùng tập trung vào việc tối ưu hóa cấu trúc lưu trữ và đảm bảo tính toàn vẹn tham chiếu cho các tác vụ phân tích:
  
  \begin{itemize}
      \item \textbf{Tối ưu hóa Engine (SQLite Tuning):}
          \begin{itemize}
              \item Kích hoạt \texttt{PRAGMA WAL} (Write-Ahead Logging) để tối ưu hiệu suất ghi dữ liệu lớn.
              \item Cấu hình \texttt{Synchronous=NORMAL} giúp tăng tốc độ xử lý I/O đĩa cứng.
          \end{itemize}
      \item \textbf{Quản lý Khóa thay thế (Surrogate Keys):}
          \begin{itemize}
              \item Tự động ánh xạ mã định danh tự nhiên (IATA) sang khóa số nguyên (\texttt{Integer keys}).
              \item Xây dựng Dictionary mapping giúp tầng Transform thay thế dữ liệu thô bằng các khóa ngoại chuẩn xác.
          \end{itemize}
      \item \textbf{Chiến lược nạp Batch (Batch Loading):}
          \begin{itemize}
              \item Chia nhỏ dữ liệu Fact thành các đợt 50,000 dòng để nạp vào Database.
              \item Cơ chế \texttt{Commit} từng giai đoạn giúp đảm bảo an toàn dữ liệu và giải phóng bộ nhớ đệm.
          \end{itemize}
  \end{itemize}

  \end{frame}
